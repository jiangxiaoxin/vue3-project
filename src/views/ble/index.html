NUS服务说明

Nordic UART Service (NUS) 模拟了传统的UART串口通信，通过BLE实现设备间的双向数据传输。

特征UUID

TX: 6e400002-b5a3-f393-e0a9-e50e24dcca9e (Notify)

RX: 6e400003-b5a3-f393-e0a9-e50e24dcca9e (Write)

注意： 要使用此演示，您需要有一个支持NUS服务的BLE设备，如Nordic开发板或某些智能设备。

代码实现

以下是如何使用Web Bluetooth API与NUS服务进行通信的完整代码：

// NUS服务UUID
const NUS_SERVICE_UUID = '6e400001-b5a3-f393-e0a9-e50e24dcca9e';
const TX_CHARACTERISTIC_UUID = '6e400002-b5a3-f393-e0a9-e50e24dcca9e'; // 设备→客户端
const RX_CHARACTERISTIC_UUID = '6e400003-b5a3-f393-e0a9-e50e24dcca9e'; // 客户端→设备

let device = null;
let server = null;
let nusService = null;
let txCharacteristic = null;
let rxCharacteristic = null;

// 连接设备
async function connectDevice() {
    try {
        // 请求设备
        device = await navigator.bluetooth.requestDevice({
            filters: [{ services: [NUS_SERVICE_UUID] }]
        });
        
        // 连接GATT服务器
        server = await device.gatt.connect();
        
        // 获取NUS服务
        nusService = await server.getPrimaryService(NUS_SERVICE_UUID);
        
        // 获取TX特征（用于接收数据）
        txCharacteristic = await nusService.getCharacteristic(TX_CHARACTERISTIC_UUID);
        
        // 获取RX特征（用于发送数据）
        rxCharacteristic = await nusService.getCharacteristic(RX_CHARACTERISTIC_UUID);
        
        // 监听断开连接
        device.addEventListener('gattserverdisconnected', onDisconnected);
        
        console.log('成功连接到NUS设备');
        updateUI(true);
        
    } catch (error) {
        console.error('连接失败:', error);
        updateStatus('连接失败: ' + error.message, 'error');
    }
}

// 启用接收通知
async function startNotifications() {
    try {
        await txCharacteristic.startNotifications();
        txCharacteristic.addEventListener('characteristicvaluechanged', handleReceivedData);
        console.log('已启用数据接收通知');
        updateStatus('已启用数据接收', 'success');
    } catch (error) {
        console.error('启用通知失败:', error);
        updateStatus('启用通知失败: ' + error.message, 'error');
    }
}

// 处理接收到的数据
function handleReceivedData(event) {
    const value = event.target.value;
    // 将接收到的数据转换为字符串
    const decoder = new TextDecoder('utf-8');
    const receivedString = decoder.decode(value);
    
    console.log('接收到数据:', receivedString);
    addToReceiveLog(receivedString);
}

// 发送数据到设备
async function sendData(data) {
    try {
        // 将字符串转换为Uint8Array
        const encoder = new TextEncoder();
        const dataArray = encoder.encode(data);
        
        // 通过RX特征发送数据
        await rxCharacteristic.writeValue(dataArray);
        
        console.log('数据发送成功:', data);
        addToSendLog(data);
        
    } catch (error) {
        console.error('发送数据失败:', error);
        updateStatus('发送失败: ' + error.message, 'error');
    }
}

// 断开连接
function disconnectDevice() {
    if (device) {
        if (device.gatt.connected) {
            device.gatt.disconnect();
        }
    }
    updateUI(false);
}

// 设备断开连接处理
function onDisconnected() {
    console.log('设备已断开连接');
    updateUI(false);
    updateStatus('设备已断开连接', 'error');
}
